### 次のステップ：`build.py` の実装

「飲茶坊さとし」のコンセプト（各フォルダー内でのカプセル化 ＋ Front Matter による分類）に沿った、汎用的な `build.py` スクリプトのベースを作成しましょう。

以下は、指定したフォルダー内（あるいは自フォルダー内）の `.md` ファイルを総なめし、Front Matter を解析するコード例です。

```python
import os
import glob
import subprocess
import frontmatter  # pip install python-frontmatter

# カプセル（フォルダー）の設定
TARGET_DIR = "./documents"  # 走査対象のフォルダー（指定可能）

def scan_and_parse_md(target_dir):
    """指定フォルダー内の全.mdファイルを走査し、Front Matterを抽出する"""
    cards = []
    
    # フォルダー内の全 .md ファイルを取得
    md_files = glob.glob(os.path.join(target_dir, "**/*.md"), recursive=True)
    
    for filepath in md_files:
        try:
            post = frontmatter.load(filepath)
            meta = post.metadata
            
            # 必須フィールドのデフォルト値（漏れ対策）
            card_data = {
                'type': meta.get('type', 'card'),
                'title': meta.get('title', os.path.basename(filepath)),
                'date': meta.get('date', '1970.0101.00'),
                'category': meta.get('category', target_dir),
                'tags': meta.get('tags', []),
                'filepath': filepath,
                'content': post.content  # 本文が必要な場合
            }
            cards.append(card_data)
        except Exception as e:
            print(f"Error parsing {filepath}: {e}")
            
    # 日付降順（最新順）に並び替え
    cards.sort(key=lambda x: str(x['date']), reverse=True)
    return cards

def update_index_html(cards, output_path):
    """取得したデータを使って index.html を書き換える処理（ダミー例）"""
    print(f"[{output_path}] に {len(cards)} 件のカードを出力します。")
    # TODO: Jinja2テンプレートや文字列置換等で index.html を生成するロジックを配置

def auto_git_commit_push(commit_message="auto: build site via build.py"):
    """GitBashの ステージング → 確定 → 昇華（Push）を自動化"""
    try:
        # 1. ステージング
        subprocess.run(["git", "add", "."], check=True)
        # 2. 確定
        subprocess.run(["git", "commit", "-m", commit_message], check=True)
        # 3. 昇華（リモートへPush）
        subprocess.run(["git", "push", "origin", "main"], check=True)
        print("Git push completed successfully.")
    except subprocess.CalledProcessError as e:
        print(f"Git command failed: {e}")

if __name__ == "__main__":
    # 1. 解析
    cards = scan_and_parse_md(TARGET_DIR)
    
    # 2. HTML生成
    index_html_path = os.path.join(TARGET_DIR, "index.html")
    update_index_html(cards, index_html_path)
```
